cmake_minimum_required(VERSION 3.20)
project(DualForm VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
include(CTest)
option(CRYPTO_BUILD_RESEARCH "Build bounded research tools" OFF)
option(CRYPTO_BUILD_BENCHMARKS "Build measurement tools" OFF)
option(CRYPTO_ENABLE_SANITIZERS "Enable ASan and UBSan" OFF)
if(BUILD_TESTING OR CRYPTO_BUILD_RESEARCH OR CRYPTO_BUILD_BENCHMARKS)
  find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
endif()
if(CRYPTO_ENABLE_SANITIZERS)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
    add_link_options(-fsanitize=address,undefined -no-pie)
  else()
    message(FATAL_ERROR "Sanitizers require Linux GCC or Clang")
  endif()
endif()
if(MSVC)
  add_compile_options(/W4)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
  add_executable(test_width tests/test_width.c)
  target_link_libraries(test_width PRIVATE OpenSSL::Crypto)
  add_test(NAME integer_width COMMAND test_width)
endif()
if(BUILD_TESTING)
  add_executable(test_primitives tests/test_primitives.c)
  target_include_directories(test_primitives PRIVATE src)
  add_test(NAME primitives COMMAND test_primitives)
  # Assertions are test checks even in Release builds.
  if(MSVC)
    target_compile_options(test_primitives PRIVATE /UNDEBUG)
  else()
    target_compile_options(test_primitives PRIVATE -UNDEBUG)
  endif()
endif()
