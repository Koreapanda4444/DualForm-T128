cmake_minimum_required(VERSION 3.20)
project(DualForm VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
include(CTest)
option(CRYPTO_BUILD_RESEARCH "Build bounded research tools" OFF)
option(CRYPTO_BUILD_BENCHMARKS "Build measurement tools" OFF)
option(CRYPTO_ENABLE_SANITIZERS "Enable ASan and UBSan" OFF)
if(BUILD_TESTING OR CRYPTO_BUILD_RESEARCH OR CRYPTO_BUILD_BENCHMARKS)
  find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
endif()
if(CRYPTO_ENABLE_SANITIZERS)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
    add_link_options(-fsanitize=address,undefined -no-pie)
  else()
    message(FATAL_ERROR "Sanitizers require Linux GCC or Clang")
  endif()
endif()
if(MSVC)
  add_compile_options(/W4)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
  add_executable(test_width tests/test_width.c)
  target_link_libraries(test_width PRIVATE OpenSSL::Crypto)
  add_test(NAME integer_width COMMAND test_width)
endif()
if(BUILD_TESTING)
  add_executable(test_primitives tests/test_primitives.c)
  target_include_directories(test_primitives PRIVATE src)
  add_test(NAME primitives COMMAND test_primitives)
  # Assertions are test checks even in Release builds.
  if(MSVC)
    target_compile_options(test_primitives PRIVATE /UNDEBUG)
  else()
    target_compile_options(test_primitives PRIVATE -UNDEBUG)
  endif()
endif()
add_library(dualform STATIC src/shake256.c src/dualform_core.c src/dualform.c)
target_link_libraries(dualform PUBLIC OpenSSL::Crypto)
target_include_directories(dualform PRIVATE src PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
function(df_test name)
  add_executable(${name} tests/${name}.c)
  target_link_libraries(${name} PRIVATE dualform)
  target_include_directories(${name} PRIVATE src include)
  if(MSVC)
    target_compile_options(${name} PRIVATE /UNDEBUG)
  else()
    target_compile_options(${name} PRIVATE -UNDEBUG)
  endif()
  add_test(NAME ${name} COMMAND ${name})
endfunction()
if(BUILD_TESTING)
  df_test(test_shake)
endif()

if(BUILD_TESTING)
  df_test(test_substitute)
endif()
if(BUILD_TESTING)
  df_test(test_rounds)
endif()

if(BUILD_TESTING)
  df_test(test_key)
endif()
if(BUILD_TESTING)
  df_test(test_tweak)
endif()
if(BUILD_TESTING)
  add_test(NAME python_reference COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/reference.py)
endif()
if(BUILD_TESTING)
  df_test(test_prepared)
endif()
if(BUILD_TESTING)
  df_test(test_encrypt)
endif()
if(BUILD_TESTING)
  add_executable(vector_runner tests/vector_runner.c)
  target_include_directories(vector_runner PRIVATE src tests)
  target_link_libraries(vector_runner PRIVATE dualform)
  add_test(NAME fixed_vectors COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_vectors.py $<TARGET_FILE:vector_runner>)
endif()
if(BUILD_TESTING)
  add_custom_target(check-differential
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_differential.py $<TARGET_FILE:vector_runner>
    DEPENDS vector_runner VERBATIM)
endif()
